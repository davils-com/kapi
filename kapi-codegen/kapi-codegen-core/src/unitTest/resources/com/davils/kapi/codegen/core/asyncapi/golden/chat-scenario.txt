private const val HOST: String = "127.0.0.1"

private const val LIMIT: Int = 256

private const val APPLICATION_CODES: Int = 4000

private object Tokens : Authorizer {
    override suspend fun authorize(attempt: AuthorizationAttempt): Authorization {
        val token = (attempt.credential as? PresentedCredential.Token)?.token?.reveal()
        return Authorization.Granted(principal("ann")).takeIf { token == "t1" } ?: Authorization.Unknown
    }
}

private class Room : ChatRoomService {
    override suspend fun serve(session: ChatRoomServerSession) {
        session.sendJoined(ChatJoinedPayload(type = "joined", member = ChatMember(session.nick)))
        session.incoming.collect { message ->
            when (message) {
                is ChatRoomClientMessage.Say -> {
                    val text = listOf(message.payload.text, session.roomId, session.protocol, session.since, session.xClient)
                    session.sendSaid(ChatSaid(type = "said", from = ChatMember(session.nick), text = text.joinToString(":")))
                }
                is ChatRoomClientMessage.Leave -> session.close(4001.toShort(), "bye")
            }
        }
    }
}

public object Scenario {
    public fun run(): List<String> = runBlocking {
        val server = embeddedServer(CIO, port = 0, host = HOST) {
            install(ServerWebSockets) { maxFrameSize = LIMIT.toLong() }
            install(Kapi) { authorizer("token", Tokens) }
            routing {
                mount(ChatRoutes(Room(), channelLimits { maxFrameBytes = LIMIT; maxMessageBytes = LIMIT }))
            }
        }
        server.start(wait = false)
        try {
            val port = server.engine.resolvedConnectors().first().port
            HttpClient(ClientEngine) {
                followRedirects = false
                install(ClientWebSockets) { maxFrameSize = ChannelSettings.Default.maxFrameBytes }
            }.use { http -> talk(clientOf(http, port, "t1")) + talk(clientOf(http, port, "wrong")).take(1) + admitted() }
        } finally {
            server.stop(gracePeriodMillis = 0, timeoutMillis = 0)
        }
    }

    private suspend fun admitted(): List<String> {
        val endpoint = ChatRoutes(Room()).channels.first()
        return listOf("since=7", "nick=ann&since=nope", "nick=ann&since=7").map { query ->
            val request = Request.of(operation = "room", method = "GET", path = "/rooms/lobby") { rawQuery = query }
            "admission " + endpoint.admission(request)?.status
        }
    }

    private suspend fun talk(client: ApiClient): List<String> {
        val room = ChatRoomChannel(client)
        val chat = room.connect(roomId = "lobby", nick = "ann", since = 7, xClient = "test") {
            val seen = mutableListOf<String>()
            sendSay(ChatSay(type = "say", text = "hi"))
            incoming.collect { message ->
                seen += shown(message)
                if (message is ChatRoomServerMessage.Said) sendLeave(ChatLeavePayload(type = "leave"))
            }
            seen.joinToString(", ") + " / " + subprotocol
        }
        val large = room.connect(roomId = "lobby", nick = "bob") {
            sendSay(ChatSay(type = "say", text = "x".repeat(LIMIT * 2)))
            incoming.collect { _ -> }
            "large"
        }
        return listOf(outcome(chat), outcome(large))
    }

    private fun clientOf(http: HttpClient, port: Int, token: String): ApiClient = apiClient(http) {
        baseUrl = "http://$HOST:$port"
        destinations = destinationPolicy {
            schemes = setOf("http")
            hosts = setOf(HOST)
        }
        credentials = credentials { scheme("token") { Secret(token) } }
    }

    private fun shown(message: ChatRoomServerMessage): String = when (message) {
        is ChatRoomServerMessage.Joined -> "joined " + message.payload.member.nick
        is ChatRoomServerMessage.Said -> "said " + message.payload.from.nick + " " + message.payload.text
    }

    private fun outcome(outcome: ChannelOutcome<String>): String = when (outcome) {
        is ChannelOutcome.Completed -> {
            val reason = (" " + outcome.close.reason).takeIf { outcome.close.code >= APPLICATION_CODES }.orEmpty()
            outcome.value + " closed " + outcome.close.code + reason
        }
        is ChannelOutcome.Failed -> "failed " + outcome.failure
    }
}
