/**
 * The WebSocket channels `room`, mounted with `routing { mount(routes) }` below the path of the server.
 *
 * Every handshake is authorized by the security requirement of its channel before the WebSocket opens. Its parameters are read after it opened: a handshake that lacks a required one or carries one that is invalid closes the channel with 1008 before the service sees it.
 *
 * @param chatRoomService serves the channel `room`.
 * @param limits the limits of every channel.
 */
@DavilsGenerated(
    by = "kapi-codegen-asyncapi",
    from = "chat.yaml#",
)
public class ChatRoutes(
    private val chatRoomService: ChatRoomService,
    private val limits: ChannelLimits = ChannelLimits.Default,
) : Routes {
    override val operations: List<Operation> = emptyList()

    override val channels: List<ChannelEndpoint<*, *>> = listOf(
        ChannelEndpoint(
            name = "room",
            path = "/rooms/{roomId}",
            receive = MessageCodec(ChatRoomClientMessageSerializer),
            send = MessageCodec(ChatRoomServerMessageSerializer),
            protocols = listOf("chat.v1"),
            security = listOf(listOf(SchemeUse("token", CredentialSource.Bearer, setOf()))),
            limits = limits,
            handler = { serveChatRoom(this) },
        ),
    )

    private suspend fun serveChatRoom(session: ChannelSession<ChatRoomClientMessage, ChatRoomServerMessage>) {
        val request = session.request
        val roomId = request.pathParameters["roomId"].orEmpty()
        val nick = request.query.firstOrNull { entry -> entry.first == "nick" }?.second
        val sinceText = request.query.firstOrNull { entry -> entry.first == "since" }?.second
        val since = sinceText?.toLongOrNull()
        val xClient = request.headers["X-Client"]
        if (nick == null || sinceText != null && since == null) return session.close(ChannelSession.POLICY_VIOLATION, "The handshake lacks a parameter or carries an invalid one")
        chatRoomService.serve(ChatRoomServerSession(session = session, roomId = roomId, nick = nick, since = since, xClient = xClient))
    }
}
