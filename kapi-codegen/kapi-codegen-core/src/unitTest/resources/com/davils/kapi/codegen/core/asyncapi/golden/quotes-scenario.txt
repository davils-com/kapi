private const val HOST: String = "127.0.0.1"

private class Desk : QuotesQuotesService {
    override suspend fun serve(session: QuotesQuotesServerSession) {
        while (true) {
            val message = session.receive() ?: return
            val quote = (message as? QuotesQuotesClientMessage.Quote)?.payload ?: continue
            if (quote.symbol == "CONFIRM") {
                val asked = QuotesConfirm(type = "confirm", request = QuotesRequest(id = "c-" + quote.request.id))
                val answer = session.requestConfirmQuote(asked, 5.seconds)
                val ack = ((answer as? ServerReply.Received<QuotesQuotesClientMessage>)?.message as? QuotesQuotesClientMessage.Ack)?.payload
                val cents = if (ack?.accepted == true) 1L else 0L
                session.sendPrice(QuotesPrice(type = "price", request = quote.request, cents = cents))
                continue
            }
            session.sendTick(QuotesTick(type = "tick", symbol = quote.symbol, cents = 1))
            if (quote.symbol != "SILENT") {
                session.sendPrice(QuotesPrice(type = "price", request = quote.request, cents = 4200))
            }
        }
    }
}

public object Scenario {
    public fun run(): List<String> = runBlocking {
        val keys = ApiKeyAuthorizer { key -> principal("desk").takeIf { key.reveal() == "k1" } }
        val server = embeddedServer(CIO, port = 0, host = HOST) {
            install(ServerWebSockets)
            install(Kapi) { authorizer("key", keys) }
            routing { mount(QuotesRoutes(Desk())) }
        }
        server.start(wait = false)
        try {
            val port = server.engine.resolvedConnectors().first().port
            HttpClient(ClientEngine) {
                followRedirects = false
                install(ClientWebSockets)
            }.use { http -> listOf(talk(clientOf(http, port, "k1")), talk(clientOf(http, port, "k2"))) }
        } finally {
            server.stop(gracePeriodMillis = 0, timeoutMillis = 0)
        }
    }

    private suspend fun talk(client: ApiClient): String {
        val outcome = QuotesQuotesChannel(client).connect {
            coroutineScope {
                val price = requestAskQuote(QuotesQuote(type = "quote", request = QuotesRequest(id = "r1"), symbol = "ACME"), 5.seconds)
                val silent = requestAskQuote(QuotesQuote(type = "quote", request = QuotesRequest(id = "r2"), symbol = "SILENT"), 300.milliseconds)
                sendQuote(QuotesQuote(type = "quote", request = QuotesRequest(id = "r3"), symbol = "CONFIRM"))
                val seen = mutableListOf<String>()
                incoming.take(4).collect { message ->
                    seen += shown(message)
                    val confirm = (message as? QuotesQuotesServerMessage.Confirm)?.payload
                    if (confirm != null) sendAck(QuotesAck(type = "ack", request = confirm.request, accepted = true))
                }
                (listOf(reply(price), reply(silent)) + seen).joinToString(", ")
            }
        }
        return when (outcome) {
            is ChannelOutcome.Completed -> outcome.value + " closed " + outcome.close.code
            is ChannelOutcome.Failed -> "failed " + outcome.failure
        }
    }

    private fun clientOf(http: HttpClient, port: Int, key: String): ApiClient = apiClient(http) {
        baseUrl = "http://$HOST:$port"
        destinations = destinationPolicy {
            schemes = setOf("http")
            hosts = setOf(HOST)
        }
        credentials = credentials { scheme("key") { Secret(key) } }
    }

    private fun reply(reply: ChannelReply<QuotesQuotesServerMessage>): String = when (reply) {
        is ChannelReply.Received -> "reply " + shown(reply.message)
        ChannelReply.TimedOut -> "timed out"
        ChannelReply.Closed -> "closed"
        ChannelReply.Refused -> "refused"
    }

    private fun shown(message: QuotesQuotesServerMessage): String = when (message) {
        is QuotesQuotesServerMessage.Price -> "price " + message.payload.request.id + " " + message.payload.cents
        is QuotesQuotesServerMessage.Tick -> "tick " + message.payload.symbol
        is QuotesQuotesServerMessage.Confirm -> "confirm " + message.payload.request.id
    }
}
