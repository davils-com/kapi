        val body = input.body(PetstoreOrderWrite.serializer(), true, listOf(MediaRange.of("application/xml")))
        if (!input.isValid || body == null) return input.refused()
        return when (val response = service.placeOrder(body = body)) {
            is PetstorePlaceOrderResponse.Created -> reply(201) {
                body(MediaType.parse("application/xml"), PetstoreOrder.serializer(), response.body)
            }
            is PetstorePlaceOrderResponse.Conflict -> reply(409)
        }
    }
